# Survival Curves using MS-Cox and IPW Case Base Models

## Generate MS-Cox Survival Curve
```{r, cache=TRUE}
sf.cox.ind <- survfit(
  model.5,
  newdata = individual.profile.data
)

S.cox <- approx(
  sf.cox.ind$time,
  sf.cox.ind$surv,
  xout = tau.grid,
  method = "constant",
  rule = 2
)$y

Lo.cox <- approx(
  sf.cox.ind$time,
  sf.cox.ind$lower,
  xout = tau.grid,
  method = "constant",
  rule = 2
)$y

Up.cox <- approx(
  sf.cox.ind$time,
  sf.cox.ind$upper,
  xout = tau.grid,
  method = "constant",
  rule = 2
)$y
```

## Generate Case Base Survival Curve
```{r, cache=TRUE, message=FALSE}
set.seed(s)
n.boot <- 200

boot.surv <- replicate(n.boot, {
  idx <- sample(nrow(data.set), replace = TRUE)
  boot.data <- data.set[idx, ]

  boot.fit <- FSH(
    Delta ~ pspline(time, df = 2) * A,
    data = boot.data,
    ratio = 10,
    weights = ipw.stabilized[idx]
  )

  ar <- absoluteRisk(
    boot.fit,
    time = tau.grid,
    newdata = individual.profile.data,
    type = "survival"
  )
  ar[, 2]
})

S.cb  <- absoluteRisk(
  model.8,
  time = tau.grid,
  newdata = individual.profile.data,
  type = "survival"
)[, 2]

Lo.cb <- apply(
  boot.surv,
  1,
  quantile,
  probs = 0.025,
  na.rm = TRUE
)

Up.cb <- apply(
  boot.surv,
  1,
  quantile,
  probs = 0.975,
  na.rm = TRUE
)
```
